🔹 SERVICE
async getProfile({ viewerId, username }) {
    const targetUser = await userRepository.findByUsername(username)
DEPENDENCY: followservice.checkFollowStatusService()// from the followservice
    if (!targetUser) {
        throw new NotFoundError("USER_NOT_FOUND")
    }

    let isFollowing = false
    let canViewFullProfile = false

    if (viewerId) {

        if (viewerId.toString() === targetUser._id.toString()) {
            canViewFullProfile = true
        } else {

            const relation = await followRepository.findRelation(
                viewerId,
                targetUser._id
            )

            if (relation?.status === "approved") {
                isFollowing = true
                canViewFullProfile = true
            }
        }
    }

    return buildProfileResponse(targetUser, canViewFullProfile, isFollowing)
}


🔹 buildProfileResponse()
function buildProfileResponse(user, canViewFull, isFollowing) {
    return {
        username: user.username,
        avatarUrl: user.profile.avatarUrl,
        followersCount: user.followersCount,
        followingCount: user.followingCount,
        ...(canViewFull && {
            bio: user.profile.bio
        }),
        ...(isFollowing && {
            isFollowing: true
        })
    }
}

If not following:
isFollowing: false

(Explicit boolean is better than undefined.)

    const userId = req.user.userId

    const updated = await profileService.updateProfile({
        userId,
        updateData: req.body
    })

    return res.status(200).json(updated)
}


🔹 SERVICE
async updateProfile({ userId, updateData }) {

    const allowedFields = ["name", "bio", "avatarUrl"]

    const sanitized = pick(updateData, allowedFields)

    if (Object.keys(sanitized).length === 0) {
        throw new ValidationError("NO_VALID_FIELDS")
    }

    const updatedUser = await userRepository.updateProfile(
        userId,
        sanitized
    )

    return {
        username: updatedUser.username,
        avatarUrl: updatedUser.profile.avatarUrl,
        bio: updatedUser.profile.bio
    }
}


🔹 REPOSITORY
async updateProfile(userId, fields) {
    return Users.findByIdAndUpdate(
        userId,
        {
            $set: {
                "profile.name": fields.name,
                "profile.bio": fields.bio,
                "profile.avatarUrl": fields.avatarUrl,
                updatedAt: new Date()
            }
        },
        { new: true }
    )
}

Single document update.
 No transaction needed.
