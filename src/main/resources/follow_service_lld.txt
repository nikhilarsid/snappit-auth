🔹 SERVICE
async createFollowRequest({ followerId, targetUsername }) {

    // 1️⃣ Resolve username → ObjectId (Profile dependency)
    const targetUser = await profileService.getUserByUsername(targetUsername)
    if (!targetUser) throw new NotFoundError("USER_NOT_FOUND")

    if (followerId.toString() === targetUser._id.toString()) {
        throw new ValidationError("CANNOT_FOLLOW_SELF")
    }

    // 2️⃣ Check existing relation
    const existing = await followRepository.findRelation(
        followerId,
        targetUser._id
    )

    if (existing?.status === "approved")
        throw new ConflictError("ALREADY_FOLLOWING")

    if (existing?.status === "pending")
        throw new ConflictError("REQUEST_ALREADY_SENT")

    const session = await startSession()
    session.startTransaction()

    try {

        if (existing?.status === "rejected") {
            await followRepository.updateStatus(
                existing._id,
                "pending",
                session
            )
        } else {
            await followRepository.create({
                followerId,
                followingId: targetUser._id,
                status: "pending",
                createdAt: new Date(),
                respondedAt: null
            }, session)
        }

        await eventRepository.create({
            type: "FOLLOW_REQUESTED",
            aggregateId: targetUser._id,
            payload: {
                followerId,
                followingId: targetUser._id
            }
        }, session)

        await session.commitTransaction()

    } catch (err) {
        await session.abortTransaction()
        throw err
    }
}

🔹 SERVICE
async approveFollow({ currentUserId, followerUsername }) {

    const follower = await profileService.getUserByUsername(followerUsername)
    if (!follower) throw new NotFoundError()

    const relation = await followRepository.findRelation(
        follower._id,
        currentUserId
    )

    if (!relation || relation.status !== "pending")
        throw new NotFoundError("NO_PENDING_REQUEST")

    const session = await startSession()
    session.startTransaction()

    try {

        await followRepository.updateStatus(
            relation._id,
            "approved",
            session
        )

        await eventRepository.create({
            type: "FOLLOW_ACCEPTED",
            aggregateId: relation._id,
            payload: {
                followerId: follower._id,
                followingId: currentUserId
            }
        }, session)

        await session.commitTransaction()

    } catch (err) {
        await session.abortTransaction()
        throw err
    }
}

🔹 SERVICE
async getFollowers({ username, limit, cursor }) {

    // 1️⃣ Username → ObjectId
    const user = await profileService.getUserByUsername(username)
    if (!user) throw new NotFoundError()

    // 2️⃣ Query follows collection
    const relations = await followRepository.getFollowers(
        user._id,
        limit,
        cursor
    )

    const followerIds = relations.map(r => r.followerId)

    // 3️⃣ ObjectIds → usernames (Profile dependency)
    const users = await profileService.getUsernamesByIds(followerIds)

    return {
        data: users,
        nextCursor: relations.length ? relations[relations.length - 1]._id : null
    }
}


🔹 REPOSITORY
async getFollowers(userId, limit, cursor) {

    const query = {
        followingId: userId,
        status: "approved"
    }

    if (cursor) {
        query._id = { $lt: new ObjectId(cursor) }
    }

    return Follows.find(query)
        .sort({ _id: -1 })
        .limit(limit)
}

Cursor-based pagination using _id.
Efficient because:
_id is indexed


followingId + status index exists
